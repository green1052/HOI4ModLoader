#
# HOI4ModLoader AI Airborne: the major powers' AI researches paratroopers and transport planes, keeps a few paratrooper
# divisions and builds transport planes, so that AI Assist's strategic paradrops (a HOI4ModLoader plugin) have something
# to drop with: real divisions, real planes, the vanilla paradrop.
#
# AI strategy values of every enabled strategy add up. These only add, and stay small: where Sheep Mod deliberately bans
# aircraft (-999: France, Italy and the USSR beside the UK/US under 175 divisions, mass-mob and Chinese AIs, minors), the
# ban still wins. Nothing of Sheep Mod or vanilla is overwritten.
#

hoimod_ai_airborne_research = {
	allowed = {
		OR = {
			original_tag = GER
			original_tag = ENG
			original_tag = USA
			original_tag = JAP
			original_tag = SOV
			original_tag = ITA
		}
	}
	enable = {
		date > 1938.1.1
		num_of_military_factories > 30
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = research_tech
		id = paratroopers
		value = 50
	}
	ai_strategy = {
		type = research_tech
		id = paratroopers2
		value = 25
	}
	ai_strategy = {
		type = research_tech
		id = bba_early_transport_plane
		value = 50
	}
	ai_strategy = {
		type = research_tech
		id = bba_improved_transport_plane
		value = 20
	}
}

hoimod_ai_airborne_units = {
	allowed = {
		OR = {
			original_tag = GER
			original_tag = ENG
			original_tag = USA
			original_tag = JAP
			original_tag = SOV
			original_tag = ITA
		}
	}
	enable = {
		has_tech = paratroopers
		num_of_military_factories > 40
	}
	abort_when_not_enabled = yes

	# About 4% of the divisions as paratroopers (vanilla's default_major_SF_para adds its own 4 with special forces).
	ai_strategy = {
		type = role_ratio
		id = paratroopers
		value = 4
	}
	# Sheep Mod pushes Germany's paratrooper template down (-100, and -50 before 1940): back to usable.
	ai_strategy = {
		type = template_prio
		id = paratroopers
		value = 120
	}
	ai_strategy = {
		type = template_prio
		id = paratrooper
		value = 120
	}
	# Transport planes: a paradrop needs 50 of them at least (NAir.MIN_PLANE_COUNT_PARADROP).
	ai_strategy = {
		type = unit_ratio
		id = air_transport
		value = 5
	}
	ai_strategy = {
		type = equipment_production_min_factories
		id = air_transport
		value = 2
	}
}
